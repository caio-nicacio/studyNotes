	\section{Curvas e parametrização}
	\label{sec:curvas}
	
	Seja $\gamma:[a, b]\to \Omega$. Uma parametrização de $\gamma$ é uma função, tal que,
	\begin{gather*}
		\boldsymbol{\vec{r}}:\Omega \subseteq \mathbb{R}\to \mathbb{R}^n \\
		\boldsymbol{\vec{r}}(t)=(x_1(t),\, x_2(t),\, \cdots,\, x_n(t)),
	\end{gather*}
	cuja imagem de $\boldsymbol{\vec{r}}$ é exatamente a curva $\gamma$ (Fig. \ref{fig:parametrizacao}). Verifica-se, portanto, que
	\begin{equation*}
		\gamma=\{\boldsymbol{\vec{r}}(t)\in \mathbb{R}^n: t\in [a,b]\}.
	\end{equation*}
	Uma mesma curva $\gamma$ pode ter infinitas parametrizações, as quais descrevem a mesma curva $\gamma$, mas cuja configuração influencia na velocidade e sentido em que é percorrida.
	\begin{figure}[h!]
		\centering
		\includegraphics[width=5.5cm]{figure/parametrizacao.jpeg}
		%\captionsetup{width=16cm}
		\caption{\centering Parametrização arbitrária $\boldsymbol{\vec{r}}(t)$ cuja imagem é a curva $\gamma(t):[a, b]$.}
		\label{fig:parametrizacao}
	\end{figure}

	As parametrizações
	\begin{gather*}
		\boldsymbol{\vec{r}}_1=(\cos t, \sin t, t)\\
		\boldsymbol{\vec{r}}_2=(\cos 2t, \sin 2t, t) \\
		\boldsymbol{\vec{r}}_3=(\cos t, \sin t, t/2) \\
		\boldsymbol{\vec{r}}_4=(\cos 2t, \sin 2t, t/2),
	\end{gather*}
	associadas às Fig. \ref{fig:curvas-circulo}a, \ref{fig:curvas-circulo}b, \ref{fig:curvas-circulo}c e \ref{fig:curvas-circulo}d, respectivamente, apresentam a mesma projeção da curva $\gamma=\{(x,y):x^2+y^2=1\}$, $\gamma:[a,b]\to \Omega$ no plano $\mathbb{R}^2$, que passam pelos mesmos pontos $(x,y)$.

	\begin{obs}
		$\gamma$ é um conjunto de pontos no espaço $\mathbb{R}^n$, em que a parametrização é uma maneira específica de descrever uma ordem para visitar esses pontos.
	\end{obs}

	Seja $\boldsymbol{\vec{r}}(\lambda(t))=(\cos \lambda(t), \sin \lambda(t))$ uma parametrização da curva $\gamma$. $\lambda(t)$ governa como a projeção circular é percorrida ao longo do eixo $z=t$. Logo, é conveniente interpretar $t$ como o tempo e $\lambda(t)$ como o coeficiente angular no plano $xy$, em função do tempo. A velocidade angular da projeção é, portanto,
	\begin{gather*}
		\omega(t)=\frac{d\lambda}{dt}(t)=\lambda'(t), \quad [\omega(t)]=\frac{\text{rad}}{\text{s}}.
	\end{gather*}

	A frequência vertical, a cada volta completa, é
	\begin{gather*}
		f=\frac{2\pi}{\lambda'(t)},\quad \lambda'(t)\in \mathbb{R}, \quad [f]=\text{s}^{-1}.
	\end{gather*}

	\section{Integral de linha de primeira espécie}
	\label{sec:integral-linha-1}
	
	Seja $f:\Omega \subset \mathbb{R}^2\to \mathbb{R}$ uma função escalar aplicada ao longo de uma curva $\gamma:[a,b]\to \mathbb{R}^2$.
	A expressão
	\begin{equation*}
		\int_{\gamma} f(x,y)\, ds
	\end{equation*}
	denota uma integral de linha de $f$ sobre $\gamma(t)$, onde $x=x(t)$ e $y=y(t)$. Imediatamente, tem-se que
	\begin{equation}
		\int_{\gamma} f(x(t),y(t))\, ds=\int_{\gamma} f(x(t),y(t))\, \sqrt{\left(\frac{dx}{dt}\right)^2+\left(\frac{dy}{dt}\right)^2}\, dt,
		\label{eq:integral-primeira-especie}
	\end{equation}
	onde $ds=\sqrt{\left(\frac{dx}{dt}\right)^2+\left(\frac{dy}{dt}\right)^2}\, dt$ é o comprimento infinitesimal de arco $\Delta s_i\to 0$ (Fig. \ref{fig:integral-primeira-especie}).
	\begin{figure}[h!]
		\centering
		\includegraphics[width=7cm]{figure/integral-primeira-especie.jpeg}
		%\captionsetup{width=16cm}
		\caption{\centering Integral de linha de uma função $f:\Omega \subset \mathbb{R}^2$ aplicada a uma curva $C\to \Omega$.}
		\label{fig:integral-primeira-especie}
	\end{figure}
	\begin{obs}
		Nota-se que, para definir a integral, não é necessário que $f$ esteja definida fora da imagem de $\gamma$.
	\end{obs}
	
	\section{Integral de linha de segunda espécie}
	\label{sec:integral-linha-2}
	
	\begin{teorema}
		Seja $\boldsymbol{\vec{F}}:\Omega \subset \mathbb{R}^2\to \mathbb{R}^2$ um campo de forças definido no aberto $\Omega$ e que uma partícula descreva um movimento em $\Omega$ com função de posição $\gamma:[a, b]\to \Omega$. $\gamma(t)$ representa a posição da partícula 
		no instante $t$.
	\end{teorema}
	
	Se $\boldsymbol{\vec{F}}$ for constante e a imagem de $\gamma$ um segmento, o trabalho $\tau$ realizado por $\boldsymbol{\vec{F}}$ de $\gamma(a)$ até $\gamma(b)$ é, por definição, o produto escalar de $\boldsymbol{\vec{F}}$ pelo deslocamento $\gamma(b)-\gamma(a)$.
	\begin{gather*}
		\tau=\boldsymbol{\vec{F}}\cdot [\gamma(b)-\gamma(a)] \\
		\tau=|\boldsymbol{\vec{F}}|\, |\gamma(b)-\gamma(a)|\, \cos \theta
	\end{gather*}
	Sejam $\boldsymbol{\vec{F}}$ contínuo e $\gamma \in C^1$. Deseja-se definir o trabalho $\tau$ realizado por $\boldsymbol{\vec{F}}$ de $\gamma(a)$ até $\gamma(b)$. Seja $P:[a,b]$ uma partição contínua do domínio, no qual o intervalo entre cada item é infinitesimal, $\Delta t_i$ suficientemente pequeno (Fig. \ref{fig:trabalho}).
	
	\begin{figure}[h!]
		\centering
		\includegraphics[width=5.5cm]{figure/trabalho.jpg}
		%\captionsetup{width=16cm}
		\caption{\centering Força $\boldsymbol{\vec{F}}(\gamma(t_{i-1}))$ aplicada a uma distância infinitesimal $\gamma(t_i)-\gamma(t_{i-1})$.}
		\label{fig:trabalho}
	\end{figure}
	
	É razoável considerar que
	\begin{equation}
		\tau=\sum_{i=1}^{n} \boldsymbol{\vec{F}}(\gamma (t_{i-1}))\cdot [\gamma(t_i)-\gamma(t_{i-1})]
		\label{eq:tau-1}
	\end{equation}
	seja uma boa aproximação para o trabalho $\tau$ realizado por $\boldsymbol{\vec{F}}$ de $\gamma(a)$ até $\gamma(b)$ e que esta aproximação seja tanto melhor quanto menor for o intervalo máx $\Delta t_i$. Assim,
	\begin{gather}
		\frac{d\gamma}{dt}(t_{i-1})=\lim_{\text{máx}\, \Delta t_i\to 0} \frac{\gamma(t_i)-\gamma(t_{i-1})}{\Delta t_i} \nonumber\\
		\frac{d\gamma}{dt}(t_{i-1})\approx \frac{\gamma(t_i)-\gamma(t_{i-1})}{\Delta t_i} \nonumber\\
		\gamma(t_i)-\gamma(t_{i-1})\approx \frac{d\gamma}{dt}(t_{i-1})\, \Delta t_i.
		\label{eq:tau-2}
	\end{gather}
	Logo, substituindo a Eq. \ref{eq:tau-2} na Eq. \ref{eq:tau-1}, é obtido
	\begin{equation*}
		\tau\approx \sum_{i=1}^{n}\boldsymbol{\vec{F}}(\gamma(t_{i-1}))\cdot \frac{d\gamma}{dt}(t_{i-1})\, \Delta t_i.
	\end{equation*}
	Como $\boldsymbol{\vec{F}}\cdot \frac{d\gamma}{dt}$ em $[a,b]$ é integrável e contínua no intervalo, segue que
	\begin{equation*}
		\lim_{\text{máx}\, \Delta t_i\to 0} \tau=\lim_{\text{máx}\, \Delta t_i\to 0} \sum_{i=1}^{n} \boldsymbol{\vec{F}}(\gamma (t_{i-1}))\cdot [\gamma(t_i)-\gamma(t_{i-1})]=\int_{a}^{b} \boldsymbol{\vec{F}}(\gamma(t))\cdot\frac{d\gamma}{dt}(t)\, dt.
	\end{equation*}
	Portanto, o trabalho realizado por $\boldsymbol{\vec{F}}$ de $\gamma(a)$ até $\gamma(b)$ pode ser definido por
	\begin{equation*}
		\tau=\int_{a}^{b} \boldsymbol{\vec{F}}(\gamma(t))\cdot\frac{d\gamma}{dt}(t)\, dt,
	\end{equation*}
	em que $\int_{a}^{b} \boldsymbol{\vec{F}}\cdot d\gamma$ é a notação para indicar a integral $\int_{a}^{b} \boldsymbol{\vec{F}}(\gamma(t))\cdot\frac{d\gamma}{dt}(t)\, dt$.
	\begin{teorema}
		Sejam $\boldsymbol{\vec{F}}:\Omega \subset \mathbb{R}^n\to \mathbb{R}^n$ e $\gamma\in C^1$. A integral de linha de $\boldsymbol{\vec{F}}$ sobre $\gamma$ é definida por
	\end{teorema}
	\begin{equation}
		\int_{a}^{b} \boldsymbol{\vec{F}}\cdot d\gamma=\int_{a}^{b} \left\langle \boldsymbol{\vec{F}}(\gamma(t)), \frac{d\gamma}{dt}(t)\, dt \right\rangle.
		\label{eq:integral-linha}
	\end{equation}
	
	\subsection{Outra notação para a integral de linha de segunda espécie}
	
	Seja $\boldsymbol{\vec{F}}(x,y)=P(x,y)\, \mathbf{i}+Q(x,y)\, \mathbf{j}$, $\boldsymbol{\vec{F}}:\Omega \subset \mathbb{R}^2\to \mathbb{R}^2$ e seja $\gamma:[a,b]\to \Omega$ uma curva de classe $C^1$, onde $x=x(t)$ e $y=y(t)$.
	\begin{align*}
		\int_{\gamma} \boldsymbol{\vec{F}}\cdot d\gamma&=\int_{a}^{b}\left[P(x,y)\, \mathbf{i}+Q(x,y)\, \mathbf{j}\right]\cdot \left[\frac{dx}{dt}\, \mathbf{i}+\frac{dy}{dt}\, \mathbf{j}\right]\, dt \\
		&=\int_{a}^{b}\left[P(x,y)\, \frac{dx}{dt}+Q(x,y)\, \frac{dy}{dt}\right]\, dt \\
		\int_{\gamma} \boldsymbol{\vec{F}}\cdot d\gamma&=\int_{a}^{b} P(x(t),y(t))\, dx + Q(x(t),y(t))\, dy
	\end{align*}
	
	\subsection{Integral de linha de segunda espécie sobre uma curva $\gamma$ por partes}
	
	Uma curva $\gamma:[a,b]\to \Omega\subseteq \mathbb{R}^n$ se diz classe $C^1$ por partes se for contínua e existirem uma partição $a=t_0<t_1<\cdots<t_n=b$ e curvas $\gamma_i:[t_{i-1},t_i]\to \Omega\subseteq \mathbb{R}^n$, $i=1,2,\cdots,n$, de classe $C^1$, tais que, $\forall t\in \left]t_{i-1}, t_i\right[$, $\gamma(t)=\gamma_i(t)$ (Fig. \ref{fig:gamma-partes}).
	
	Seja $\boldsymbol{\vec{F}}:\Omega \subset \mathbb{R}^n \to \mathbb{R}^n$ e seja $\gamma:[a,b]\to \Omega$ uma curva de classe $C^1$ por partes, tem-se
	\begin{gather*}
		\int_{\gamma} \boldsymbol{\vec{F}}\cdot d\gamma=\int_{\gamma_1} \boldsymbol{\vec{F}}\cdot d\gamma_1 + \int_{\gamma_2} \boldsymbol{\vec{F}}\cdot d\gamma_2 + \cdots + \int_{\gamma_{n-1}} \boldsymbol{\vec{F}}\cdot d\gamma_{n-1} + \int_{\gamma_n} \boldsymbol{\vec{F}}\cdot d\gamma_n \\
		\int_{\gamma} \boldsymbol{\vec{F}}\cdot d\gamma=\sum_{i=1}^{n}\int_{\gamma_i} \boldsymbol{\vec{F}}\cdot d\gamma_i.
	\end{gather*}

	\subsection{Mudança de parâmetro}
	
	Sejam $\gamma_1:[a,b]\to \Omega$ e $\gamma_2:[c,d]\to \Omega$ duas curvas classe $C^1$. Admitindo a existência de uma função $g:[c,d]\to \mathbb{R}$, $g\in C^1$, com $g'(u)>0$ em $]c,d[$ em $\text{Im}\, g=[a,b]$, tal que, $\forall u\in [c,d]$,
	\begin{equation*}
		\gamma_2(u)=\gamma_1(g(u))=\gamma_1(t),\quad t=g(u).
	\end{equation*}
	Portanto, $\gamma_2$ é obtida de $\gamma_1$ por uma \textbf{mudança de parâmetro que conserva a orientação}\footnote{O domínio de $\gamma_1$ é função do domínio de $\gamma_2$.}. Assim, $\exists g:D_{\gamma_2}\to D_{\gamma_1}$ de modo a igualar as duas curvas.
	
	Nota-se que, se $\gamma_2$ é obtida por uma mudança de parâmetro que conserva a orientação, então $\gamma_1$ e $\gamma_2$ têm a mesma imagem. Além disso, visto que
	\begin{gather*}
		\gamma_2'(u)=\frac{d\gamma_1}{dg}\, \frac{dg}{du}=\gamma_1'(t) \\
		\gamma_2'(u)=\gamma_1'(g(u))\cdot g'(u),\quad g'(u)>0,
	\end{gather*}
	segue que $\gamma_1'(t)$ e $\gamma_2'(u)$ são paralelos e terão mesmo sentido (Fig. \ref{fig:mudanca-parametro}). 
	
	\begin{figure}[h!]
		\centering
		\includegraphics[width=9.5cm]{figure/mudanca-parametro.jpeg}
		%\captionsetup{width=16cm}
		\caption{\centering Vetor tangente às curvas $\gamma_1(t)$ e $\gamma_2(u)$ no ponto $t=g(u)$.}
		\label{fig:mudanca-parametro}
	\end{figure}
	
	\begin{obs}
		Para curvas fechadas e periódicas, não importa a quantidade de vezes em que a imagem é percorrida, visto que a curva é apenas um conjunto de pontos. A parametrização é que define o modo em que a curva é descrita\footnote{No entanto, para casos gerais, a distância percorrida (tamanho do domínio) deve ser a mesma.}.
	\end{obs}

	Se a condição $g'(u)>0$ em $]c,d[$ for substituída por $g'(u)<0$ em $]c,d[$, então $\gamma_2$ é obtida por uma \textbf{mudança de parâmetro que reverte a orientação}.
	\begin{gather*}
		g'(u)>0:\qquad \text{$t$ cresce segundo $u$} \\
		g'(u)<0:\qquad \text{$t$ decresce segundo $u$}
	\end{gather*}
	O vetor tangente na Fig. \ref{fig:mudanca-parametro} pode representar a velocidade: quanto maior $g'(u)$, maior será a velocidade de percurso em $\gamma_2$. Logo, $g(u)$ é uma função usada para reparametrizar uma curva, quando necessário\footnote{Tal como $\lambda$, visto na Seção \ref{sec:curvas}.}.
	\begin{teorema}
		Seja $\boldsymbol{\vec{F}}:\Omega \subset \mathbb{R}^n\to \mathbb{R}^n$ um campo vetorial e sejam $\gamma_1:[a,b]\to \Omega$ e $\gamma_2:[c,d]\to \Omega$ duas curvas de classe $C^1$.
	\end{teorema}
	\begin{enumerate}
		\item Se $\gamma_2$ for obtida de $\gamma_1$ por uma \textbf{mudança de parâmetro que conserva a orientação},
		\begin{equation*}
			\int_{\gamma_2} \boldsymbol{\vec{F}}\cdot d\gamma_2=\int_{\gamma_1} \boldsymbol{\vec{F}}\cdot d\gamma_1.
		\end{equation*}
	
		\item Se $\gamma_2$ for obtida de $\gamma_1$ por uma \textbf{mudança de parâmetro que reverte a orientação},
		\begin{equation*}
			\int_{\gamma_2} \boldsymbol{\vec{F}}\cdot d\gamma_2=-\int_{\gamma_1} \boldsymbol{\vec{F}}\cdot d\gamma_1.
		\end{equation*}
	\end{enumerate}

	\begin{exemplo}
		Seja $\gamma_1(t)=(\cos t, \sin t)$, $t\in [0,2\pi]$. Define-se $g(u)=2u$, $u\in [0,\pi]$.
	\end{exemplo}
	$\gamma_1$ descreve uma curva segundo um parâmetro $t$. A função $g:D_{\gamma_2}\to D_{\gamma_1}$ altera o parâmetro, associando a cada $u \in [0,\pi]$, um valor $t=g(u)$.
	\begin{gather*}
		\gamma_2(u)=\gamma_1(g(u))=\gamma_1(t) \\
		\gamma_2(u)=(\cos 2u, \sin 2u)
	\end{gather*}
	$\gamma_2(u)$ é uma nova parametrização, com a mesma imagem, mas com o \textbf{dobro de velocidade}\footnote{Mapeando os valores de $t$ e $u$, sob o mesmo incremento $\Delta=t_{i+1}-t_{i}=u_{k+1}-u_{k}$, $\gamma_2(u)=\gamma_1(g(u))$ será mais rápido que $\gamma_1(t)$.}.
	
	Para que a mudança de parâmetro seja válida, a relação
	\begin{equation*}
		\gamma_2'(u)=\gamma_1'(g(u))\cdot g'(u)
	\end{equation*}
	deve ser satisfeita. Neste caso, $\gamma_1(g(u))$ é $g'(u)$ vezes mais rápido que $\gamma_1(t)$ sob o mesmo incremento $\Delta$.
	
	O propósito da mudança de parâmetro reside no fato de que, em vez de $t$ ser a variável independente, o seu crescimento irá depender de outro parâmetro, cujo incremento irá contribuir para o incremento de $t$, $\Delta t=g(u_{i+1})-g(u_i)$\footnote{Se $\Delta u$ for suficientemente pequeno, $\Delta t\approx g'(u)\, \Delta u$.}.
	
	\begin{exemplo}
		Seja $\boldsymbol{\vec{F}}:\Omega \subset \mathbb{R}^2\to \mathbb{R}^2$ e sejam $\gamma_1(t)=(t,t^2)$, $t\in [0,1]$ e $\gamma_2(u)=\left(\frac{u}{2},\frac{u^2}{4}\right)$, $u\in [0,2]$ curvas parametrizadas com mesma imagem.
	\end{exemplo}
	Considere a igualdade
	\begin{equation*}
		\int_{\gamma_2}\boldsymbol{\vec{F}}\cdot d\gamma_2=\int_{\gamma_1}\boldsymbol{\vec{F}}\cdot d\gamma_1
	\end{equation*}
	e justifique-a.
	
	Para que a igualdade seja satisfeita, ambas as curvas devem ser de classe $C^1$ e suas derivadas contínuas no domínio, assim como $g:[c,d]\to \mathbb{R}$, $g':]c,d[\to \mathbb{R}$.
	\begin{gather*}
		\frac{d\gamma_1}{dt}=(1, 2t),\quad \frac{d\gamma_2}{du}=\left(\frac{1}{2}, \frac{u}{2}\right) \\
		t=g(u)=\frac{u}{2},\quad t'=g'(u)=\frac{1}{2}
	\end{gather*}
	Como $\gamma_2'(u)=\gamma_1'(g(u))\cdot g'(u)$ deve ser satisfeita, segue que
	\begin{align*}
		\left(\frac{1}{2}, \frac{u}{2}\right)&=\frac{1}{2}\cdot (1,2t) \\
		\left(\frac{1}{2}, \frac{u}{2}\right)&=\left(\frac{1}{2}, t\right)\, \longrightarrow\, \text{a igualdade é satisfeita.}
	\end{align*}
	Além disso, por definição,
	\begin{gather*}
		\int_{\gamma_1}\boldsymbol{\vec{F}}(\gamma_1(t))\cdot \gamma_1'(t)\, dt=	\int_{\gamma_1}\boldsymbol{\vec{F}}(\gamma_1(g(u)))\cdot [\gamma_1'(g(u))\cdot g'(u)]\, dt \\
		\int_{\gamma_1}\boldsymbol{\vec{F}}(\gamma_1(t))\cdot \gamma_1'(t)\, dt=\int_{\gamma_2}\boldsymbol{\vec{F}}(\gamma_2(u))\cdot \gamma_2'(u)\, dt.
	\end{gather*}
