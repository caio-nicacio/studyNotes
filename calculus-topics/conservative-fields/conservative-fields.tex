	\section{Teorema fundamental do cálculo para integrais de linha}
	\label{sec:tfc}
	
	Seja $f:[a,b]\to \mathbb{R}$, $f\in C^1$, uma função diferenciável em $]a,b[$.
	O primeiro teorema fundamental do cálculo estabelece que
	\begin{equation*}
		\int_{a}^{b} f(x)\, dx=F(b)-F(a),
	\end{equation*}
	onde $F$ é uma primitiva de $f$.
	
	Assim, a integral no intervalo $[a,b]$ é obtida avaliando um das primitivas nos extremos do intervalo.
	
	\begin{definicao} \textbf{(Conjunto conexo)}
		Um conjunto aberto $\Omega\subset \mathbb{R}^n$ é simplesmente conexo, se existe uma curva suave $\gamma:I \to \Omega$ por partes ligado a dois pontos quaisquer de $\Omega$. Logo, $\forall \boldsymbol{a}, \boldsymbol{b}  \in \Omega$, $\exists \gamma:[a,b] \to \Omega \subset \mathbb{R}^n$, tal que $\boldsymbol{a}=\gamma(a)$ e $\boldsymbol{b}=\gamma(b)$.
	\end{definicao}
	\begin{definicao} \textbf{(Conjunto desconexo)}
		Um conjunto aberto $\Omega \subset \mathbb{R}^n$ é dito desconexo se $\Omega$ é a união de dois ou mais conjuntos abertos disjuntos e não vazios, ou seja,
		\begin{equation*}
			\Omega=\bigcup_{i \in \Gamma}A_i
		\end{equation*}
		com $A_i\cap A_j = \emptyset$, $i \neq j$, em que $\Gamma$ denota um conjunto de índices e $A_i\neq \emptyset$ é aberto $\forall i \in \Gamma$.
	\end{definicao}
	\begin{definicao}
		A integral de linha\footnote{Para informações complementares, clique \href{https://github.com/caio-nicacio/studyNotes/blob/main/calculus-topics/line-integrals/line-integrals.pdf}{aqui}.} $\int_{\gamma} \vec{\boldsymbol{F}}\cdot d\gamma$, $\vec{\boldsymbol{F}}:\Omega \subset \mathbb{R}^n\to \mathbb{R}^n$ contínuo no conjunto aberto conexo $\Omega$ é independente do caminho se
		\begin{equation*}
			\int_{\gamma_1} \vec{\boldsymbol{F}}\cdot d\gamma_1=\int_{\gamma_2} \vec{\boldsymbol{F}}\cdot d\gamma_2=\dots=\int_{\gamma_k} \vec{\boldsymbol{F}}\cdot d\gamma_k,\quad i=1,2,\dots, k,
		\end{equation*}
		$\forall \gamma_i:I_i$, $\gamma_i$ fechada.
	\end{definicao}
	\begin{teorema}
		A integral de linha $\int_{\gamma}\vec{\boldsymbol{F}}\cdot d\gamma$ de um campo vetorial contínuo $\vec{\boldsymbol{F}}:\Omega\subset \mathbb{R}^n\to \mathbb{R}^n$ definido sobre um conjunto simplesmente conexo $\Omega$ é independente do caminho se, e somente se, $\int_{\gamma}\vec{\boldsymbol{F}}\cdot d\gamma=0$, $\gamma:[a,b]\to \Omega$, $\gamma$ curva fechada.
	\end{teorema}
	\begin{teorema} \label{thm:tfc2} \textbf{(Segundo teorema fundamental do cálculo)}
		Seja $\varphi:\Omega\to \mathbb{R}$ uma função potencial de $\vec{\boldsymbol{F}}$, tal que $\nabla \varphi$ é contínuo em um conjunto aberto simplesmente conexo $\Omega\subset \mathbb{R}^n$. Dada uma curva suave por partes $\gamma:[a,b]\to \Omega$, tem-se que
		\begin{equation*}
			\int_{\gamma}\nabla \varphi\cdot d\gamma = \varphi(\gamma(b))-\varphi(\gamma(b)).
		\end{equation*}
	\end{teorema}
	Seja $\gamma:[a,b]\to \Omega$ uma curva suave e seja $\varphi:\Omega\to \mathbb{R}$ função potencial de $\vec{\boldsymbol{F}}$. Define-se a função composta $g(t)=\varphi(\gamma(t))$, $\forall t\in [a,b]$.
	
	Pela regra da cadeia, $g'(t)=\nabla \varphi(\gamma(t))\cdot \gamma'(t)$. Além disso, visto que $\gamma$ é suave e $\nabla \varphi$ contínuo, $g'(t)$ é continua.
	
	Aplicando o Teorema \ref{thm:tfc2} à função  $g$, conclui-se que
	\begin{gather*}
		\int_{\gamma}\nabla \varphi\cdot d\gamma=\int_{a}^{b} \nabla \varphi(\gamma(t))\cdot \gamma'(t)\, dt=\int_{a}^{b} g'(t)\, dt\\
		\int_{\gamma}\nabla \varphi\cdot d\gamma=g(b)-g(a)\\
		\int_{\gamma}\nabla \varphi\cdot d\gamma=\varphi(\gamma(b))-\varphi(\gamma(a)).
	\end{gather*}
	Na condição de $\gamma:[a,b]\to \Omega$ ser uma curva suave por partes, particiona-se $[a,b]$ em $k$ subintervalos $[t_{i-1},t_i]$, $i=1,2,\dots,k$, e aplica-se o resultado anterior a cada segmento.
	\begin{align*}
		\int_{\gamma} \nabla \varphi\cdot d\gamma&=\int_{a}^{b} \nabla \varphi(\gamma(t)\cdot \gamma'(t)\, dt)\\
		&=\sum_{i=1}^{n}\int_{t_{i-1}}^{t_i}\nabla \varphi(\gamma(t))\cdot \gamma'(t)\, dt\\
		&=\sum_{i=1}^{n}[\varphi(\gamma(t_i))-\varphi(\gamma(t_{i-1}))]\\
		&=\varphi(\gamma(t_k))-\varphi(\gamma(t_0))\\
		\int_{\gamma} \nabla \varphi\cdot d\gamma&=\varphi(\gamma(b))-\varphi(\gamma(a)).
	\end{align*}

	\section{Campos conservativos}
	
	Um campo vetorial $\vec{\boldsymbol{F}}:\Omega \subset \mathbb{R}^n \to \mathbb{R}^n$ denomina-se conservativo se existe um campo escalar diferenciável $\varphi:\Omega \to \mathbb{R}$ tal que
	\begin{gather*}
		\nabla \varphi=\vec{\boldsymbol{F}}\\
		\frac{\partial \varphi}{\partial x}\,\mathbf{i}+\frac{\partial \varphi}{\partial y}\,\mathbf{j}+\frac{\partial \varphi}{\partial z}\,\mathbf{k}=P(x,y,z)\,\mathbf{i}+Q(x,y,z)\,\mathbf{j}+R(x,y,z)\mathbf{k}.
	\end{gather*}

	\subsection{Função potencial}
	\begin{quote}
		Função escalar cujo gradiente gera um campo vetorial, o qual formaliza a proposição de que o campo vetorial pode ser descrito como a variação espacial de uma grandeza escalar.
	\end{quote}
	Seja $\Omega \subset \mathbb{R}^n$ um domínio aberto e seja $\vec{\boldsymbol{F}}:\Omega\subset \mathbb{R}^n \to \mathbb{R}^n$ um campo vetorial contínuo. Diz-se que $\varphi:\Omega\to \mathbb{R}$ é uma função potencial de $\vec{\boldsymbol{F}}$ se $\nabla \varphi=\vec{\boldsymbol{F}}$. Isto é,
	\begin{equation*}
		\vec{\boldsymbol{F}}(x_1,x_2,\dots,x_n)=\left(\frac{\partial \varphi}{\partial x_1},\frac{\partial \varphi}{\partial x_2},\dots,\frac{\partial \varphi}{\partial x_n}\right).
	\end{equation*}
	Nesse caso, $\vec{\boldsymbol{F}}$ é denominado campo conservativo.
	
	Consequentemente, se $\vec{\boldsymbol{F}}=\nabla \varphi$, então a integral de linha entre dois pontos $A$ e $B$ é independente do caminho. Logo,
	\begin{equation*}
		\int_{\gamma}\vec{\boldsymbol{F}}\cdot d\vec{\boldsymbol{r}}=\varphi(B)-\varphi(A),
	\end{equation*}
	$\forall\gamma$ suave ligando $A$ e $B$.
	
	Em domínios simplesmente conexos, $\vec{\boldsymbol{F}}=\nabla \varphi$ se, e somente se, $\operatorname{rot} \vec{\boldsymbol{F}}=0$.
	
	\begin{teorema}\textbf{(Condição necessária, mas não suficiente)}
		Seja $\vec{\boldsymbol{F}}:\Omega \subset \mathbb{R}^n\to \mathbb{R}^n$, $n=2,3$, um campo vetorial de classe $C^1$ no aberto $\Omega$. Uma condição necessária (mas não suficiente) para $\vec{\boldsymbol{F}}$ ser considerado conservativo é
		\begin{equation*}
			\operatorname{rot}\vec{\boldsymbol{F}}=0\, \text{em}\, \Omega.
		\end{equation*} 
	\end{teorema}
	Seja $\vec{\boldsymbol{F}}:\Omega\to \mathbb{R}^3$ um campo conservativo, $\exists\varphi:\Omega\to \mathbb{R}$ tal que $\nabla\varphi=\vec{\boldsymbol{F}}$ em $\Omega$, e
	\begin{equation*}
		\frac{\partial \varphi}{\partial x}=P(x,y,z),\, \frac{\partial \varphi}{\partial y}=Q(x,y,z),\, \frac{\partial \varphi}{\partial z}=R(x,y,z).
	\end{equation*}
	Visto que $\vec{\boldsymbol{F}}\in C^1$, resulta que $\varphi\in C^2$. Logo,
	\begin{equation*}
		\frac{\partial^2 \varphi}{\partial x \partial y}=\frac{\partial^2 \varphi}{\partial y \partial x}\quad \text{(Teorema de Schwarz)}.
	\end{equation*}
	Segue que
	\begin{equation*}
		\frac{\partial P}{\partial y}=\frac{\partial Q}{\partial x}\quad \longrightarrow\quad \frac{\partial P}{\partial y}-\frac{\partial Q}{\partial x}=0.
	\end{equation*}
	Analogamente,
	\begin{equation*}
		\frac{\partial P}{\partial y}=\frac{\partial R}{\partial x}\quad \text{e}\quad \frac{\partial Q}{\partial z}=\frac{\partial R}{\partial y},
	\end{equation*}
	em $\Omega$. Logo, $\operatorname{rot}\vec{\boldsymbol{F}}=0$, $\vec{\boldsymbol{F}}:\Omega \subset \mathbb{R}^3$.
	
	\begin{teorema}\textbf{(Condição necessária e suficiente)}\label{thm:condicao-suficiente}
		Seja $\Omega\subset \mathbb{R}^n$ um conjunto aberto conexo e seja $\vec{\boldsymbol{F}}:\Omega\subset \mathbb{R}^n\to \mathbb{R}^n$. Então,
		\begin{equation*}
			\operatorname{rot}\vec{\boldsymbol{F}}=0\,\text{em}\,\Omega \iff \exists \varphi:\Omega\to \mathbb{R}\,|\,\vec{\boldsymbol{F}}=\nabla \varphi.
		\end{equation*}
		Logo, é equivalente afirmar que
		\begin{enumerate}
			\item $\vec{\boldsymbol{F}}$ é conservativo;
			\item a integral de linha de $\vec{\boldsymbol{F}}$ em $\Omega$ independe do caminho $\gamma$;
			\item $\oint_{\gamma}\vec{\boldsymbol{F}}\cdot d\vec{\boldsymbol{r}}=0$, $\forall\gamma\subset \Omega$.
		\end{enumerate}
	\end{teorema}
	\begin{obs}
		A condição $\operatorname{rot}\vec{\boldsymbol{F}}=0$ não é suficiente se $\Omega$ não for simplesmente conexo. Existem campos irrotacionais definidos em domínios com descontinuidades que não admitem potencial global\footnote{Função escalar definida em todo o domínio $\Omega$ do campo vetorial, cuja derivada reproduz o campo em todos os pontos do domínio.}.
	\end{obs}
	\begin{obs}
		A condição é topológica, não dependendo apenas do campo $\vec{\boldsymbol{F}}$, mas do conjunto $\Omega$, no qual deve ser simplesmente conexo para que $\oint_{\gamma} \vec{\boldsymbol{F}}\cdot d\gamma=0$.
	\end{obs}
	\begin{obs}
		Em um domínio $\Omega$ simplesmente conexo, um campo irrotacional $\vec{\boldsymbol{F}}$ é conservativo somente se for satisfeito
		\begin{equation*}
			\oint_{\gamma} \vec{\boldsymbol{F}}\cdot d\vec{\boldsymbol{r}}=0,\quad \forall \gamma \subset \Omega.
		\end{equation*}
		Caso $\Omega$ seja um conjunto desconexo, não há garantias de que $\vec{\boldsymbol{F}}$ seja conservativo em seu domínio, mesmo que $\operatorname{rot}\vec{\boldsymbol{F}}=0$. Apenas é seguro afirmá-lo, se $\Omega$ for simplesmente conexo e houver potencial escalar $\varphi$ global\footnote{Um potencial global é uma função escalar definida em todo o domínio $\Omega$ do campo vetorial, cuja derivada reproduz o campo em todos os pontos do domínio.}.
	\end{obs}

	\subsection{Forma diferencial exata}
	
	Seja $\vec{\boldsymbol{F}}(x,y)=P(x,y)\,\mathbf{i}+Q(x,y)\,\mathbf{j}$ definido no aberto $\Omega$.
	\begin{gather*}
		\int_{\gamma}\vec{\boldsymbol{F}}\cdot d\gamma=\int_{a}^{b}[P(x,y)\,\mathbf{i}+Q(x,y)\,\mathbf{j}]\cdot \left[\frac{dx}{dt}\,\mathbf{i}+\frac{dy}{dt}\,\mathbf{j}\right]\, dt\\
		\int_{\gamma}\vec{\boldsymbol{F}}\cdot d\gamma=\int_{a}^{b}P(x,y)\,dx+Q(x,y)\,dy
	\end{gather*}
	é uma notação para a integral de $\vec{\boldsymbol{F}}$ sobre $\gamma$.
	\begin{equation*}
		\int_{a}^{b}P(x,y)\,dx+Q(x,y)\,dy
	\end{equation*}
	denota uma forma diferencial exata se $\exists\varphi:\Omega\to \mathbb{R}$ é diferenciável tal que
	\begin{equation*}
		\frac{\partial \varphi}{\partial x}=P(x,y)\quad \text{e}\quad \frac{\partial \varphi}{\partial y}=Q(x,y).
	\end{equation*}
	A diferencial de uma dada função$\varphi:\Omega\subset \mathbb{R}^2\to \mathbb{R}$ em dado ponto $(x,y)\in \Omega$ é dada por
	\begin{equation*}
		d\varphi=\frac{\partial \varphi}{\partial x}(x,y)\, dx+\frac{\partial \varphi}{\partial y}(x,y)\, dy,
	\end{equation*}
	onde $\varphi$ denomina-se primitiva de $P(x,y)\,dx+Q(x,y)\,dy$.
	
	A forma diferencial implica na existência de um campo escalar diferenciável $\varphi:\Omega\to \mathbb{R}$ tal que, $\forall(x,y)\in \Omega$, a diferencial de $\varphi$ é dada por
	\begin{equation}
		d\varphi=P(x,y)\,dx+Q(x,y)\,dy
	\end{equation}
	\begin{teorema}
		O campo $\vec{\boldsymbol{F}}:\Omega\subset \mathbb{R}^2\to \mathbb{R}^2$ é conservativo se, e somente se, $\exists\varphi:\Omega \to \mathbb{R}$ tal que
		\begin{equation*}
			d\varphi=P(x,y)\,dx+Q(x,y)\,dy.
		\end{equation*}
	\end{teorema}
	\begin{obs}
		A proposição pode ser generalizada para $\vec{\boldsymbol{F}}:\Omega\subset \mathbb{R}^n\to \mathbb{R}^n$, se satisfeitas as condições.
	\end{obs}
	\subsection{Integral de linha de um campo conservativo}
	
	\begin{teorema}
		Seja $\vec{\boldsymbol{F}}:\Omega\subset \mathbb{R}^n\to \mathbb{R}^n$ um campo vetorial contínuo e conservativo, e seja $\varphi:\Omega\to \mathbb{R}$ uma função potencial para $\vec{\boldsymbol{F}}$ e seja $\gamma:[a,b]\to \Omega$ classe $C^1$. Então,
		\begin{equation*}
			\int_{\gamma}\vec{\boldsymbol{F}}\cdot d\gamma=\int_{\gamma}\nabla \varphi\cdot d\gamma=\varphi(\gamma(b))-\varphi(\gamma(a)).
		\end{equation*}
	\end{teorema}

	De fato, se $\vec{\boldsymbol{F}}$ é contínuo, $\varphi\in C^1$.
	\begin{equation*}
		\frac{d\varphi}{dt}(\gamma(t))=\nabla\varphi(\gamma(t))\cdot \gamma'(t)=\vec{\boldsymbol{F}}(\gamma(t))\cdot \gamma'(t)
	\end{equation*}
	Logo,
	\begin{gather*}
		\int_{\gamma}\vec{\boldsymbol{F}}\cdot d\gamma=\int_{\gamma}\vec{\boldsymbol{F}}(\gamma(t))\cdot \gamma'(t)\, dt=\int_{a}^{b}\frac{d\varphi}{dt}(\gamma(t))\, dt\\
		\int_{\gamma}\vec{\boldsymbol{F}}\cdot d\gamma=\int_{a}^{b}d\varphi(\gamma(t))=\varphi(\gamma(t))\Big|_{a}^{b}\\
		\int_{\gamma}\vec{\boldsymbol{F}}\cdot d\gamma=\varphi(\gamma(b))-\varphi(\gamma(a)).
	\end{gather*}
	Ao calcular a integral de linha, é adequado verificar se o campo vetorial é conservativo ou se a forma diferencial é exata.
	
	\begin{exemplo}
		Calcule $\int_{\gamma}x\,dx+y\,dy$, onde $\gamma=\{(x,y)=(\arctan t, \arcsin t^3)\,|\, t\in [0,1]\}$.
		
		A forma diferencial $\omega=P(x,y)\, dx+Q(x,y)\, dy$ é dita exata em $\Omega$, se $\exists\varphi:\Omega\to \mathbb{R}$, tal que
		\begin{equation*}
			d\varphi=\frac{\partial \varphi}{\partial x}\, dx+\frac{\partial \varphi}{\partial y}\, dy.
		\end{equation*}
		Se $P,Q\in C^1$ em $\Omega$, e $\Omega$ simplesmente conexo, então $\omega$ é exato se, e somente se,
		\begin{equation*}
			\frac{\partial P}{\partial y}=\frac{\partial Q}{\partial x}.
		\end{equation*}
		Verificando a condição de irrotacionalidade,
		\begin{equation*}
			\frac{\partial P}{\partial y}=\frac{\partial}{\partial y}\,x=0=\frac{\partial}{\partial x}\, y=\frac{\partial Q}{\partial x}.
		\end{equation*}
		Logo, $\omega$ é exata em $\Omega\subset \mathbb{R}^2$.
		Da expressão, tem-se que
		\begin{equation*}
			\frac{\partial \varphi}{\varphi x}=x.
		\end{equation*}
		Assim,
		\begin{equation*}
			\varphi(x,y)=\int x\,dx + g(y)=\frac{x^2}{2} + g(y).
		\end{equation*}
		Derivando $\varphi$ em relação a $y$, tal que $\frac{\partial \varphi}{\partial y}=Q(x,y)$,
		\begin{align*}
			\frac{\partial}{\partial y}\, \frac{x^2}{2}+g'(y)=y \quad\longrightarrow\quad g'(y)&=y\\
			\int g'(y)\, dy&=\int y\, dy\\
			g(y)&=\frac{y^2}{2}+C.
		\end{align*}
		Logo, 
		\begin{equation*}
			\varphi=\frac{x^2+y^2}{2}+C.
		\end{equation*}
		Portanto,
		\begin{equation*}
			\int_{\gamma}x\,dx+y\,dy=\int_{\gamma}d\varphi=\int_{\gamma(0)=(0,0)}^{\gamma(1)=\left(\frac{\pi}{4},\sin 1\right)}d\left(\frac{x^2+y^2}{2}+C\right)=\frac{x^2+y^2}{2}\Big|_{\gamma(0)=(0,0)}^{\gamma(1)=\left(\frac{\pi}{4},\sin 1\right)}.
		\end{equation*}
		Ou seja,
		\begin{equation*}
			\int_{\gamma}x\,dx+y\,dy=\frac{\pi^2+16\, \sin^2 1}{32}.
		\end{equation*}
	\end{exemplo}
	
	\begin{exemplo}
		Calcule $\int_{\gamma} \vec{\boldsymbol{F}}\cdot d\gamma$, onde
		\begin{equation*}
			\vec{\boldsymbol{F}}(x,y)=\frac{x}{x^2+y^2}\, \mathbf{i}+\frac{y}{x^2+y^2}\,\mathbf{j}
		\end{equation*}
		e $\gamma:[a,b]\to \mathbb{R}^2\setminus \{(0,0)\}$ é uma curva $C^1$ por partes e fechada.
		\begin{gather*}
			\int_{\gamma} \vec{\boldsymbol{F}}\cdot d\gamma=\int_{\gamma}\left[P(x,y)\,\mathbf{i}+Q(x,y)\,\mathbf{j}\right]\cdot \left[\frac{dx}{dt}\,\mathbf{i}+\frac{dy}{dt}\,\mathbf{j}\right]\, dt\\
			\int_{\gamma} \vec{\boldsymbol{F}}\cdot d\gamma=\int_{\gamma}P(x,y)\, dx+Q(x,y)\, dy
		\end{gather*}
		Verificando a condição de irrotacionalidade, tem-se que
		\begin{gather*}
			\frac{\partial P}{\partial y}=\frac{\partial}{\partial y}\left(\frac{x}{x^2+y^2}\right)=\frac{2xy}{(x^2+y^2)^2}\\
			\frac{\partial Q}{\partial x}=\frac{\partial}{\partial x}\left(\frac{y}{x^2+y^2}\right)=\frac{2xy}{(x^2+y^2)^2}\\
			\frac{\partial P}{\partial y}=\frac{\partial Q}{\partial x},
			\quad\operatorname{rot}\vec{\boldsymbol{F}}=0
		\end{gather*}
		Se $\vec{\boldsymbol{F}}$ é conservativo, $\exists\varphi:\Omega\to \mathbb{R}$, tal que $\nabla\varphi=\vec{\boldsymbol{F}}$.
		\begin{align*}
			\frac{\partial \varphi}{\partial x}=P(x,y) \quad\longrightarrow\quad \varphi(x,y)&=\int \frac{x}{x^2+y^2}\,dx + g(y)\\
			&=\frac{1}{2}\int\frac{du}{u} + g(y),\quad u=x^2+y^2,\quad \frac{du}{u}=x\,dx\\
			\varphi(x,y)&=\frac{1}{2}\ln |x^2+y^2| + g(y)
		\end{align*}
		Derivando $\varphi$ em relação a y, tem-se
		\begin{align*}
			\frac{\partial \varphi}{\partial y}=Q(x,y) \quad\longrightarrow\quad \frac{\partial}{\partial y}\left(\frac{1}{2}\ln |x^2+y^2|\right)+g'(y)&=\frac{y}{x^2+y^2}\\
			\frac{1}{2}\cdot\frac{\partial}{\partial}\ln|u|\cdot\frac{\partial}{\partial y}(x^2+y^2)+g'(y)&=\frac{y}{x^2+y^2}, \quad u=x^2+y^2\\
			\frac{1}{2(x^2+y^2)}\,2y+g'(y)&=\frac{y}{x^2+y^2}\\
			g'(y)&=\frac{y}{x^2+y^2}-\frac{y}{x^2+y^2}\\
			g'(y)&=0 \quad\longrightarrow\quad g(y)=0
		\end{align*}
		Assim,
		\begin{equation*}
			\varphi(x,y)=\frac{1}{2}\ln |x^2+y^2|+C.
		\end{equation*}
		Visto que $\vec{\boldsymbol{F}}$ é conservativo e $\exists\varphi:\Omega \to \mathbb{R}$, tal que $\nabla\varphi=\vec{\boldsymbol{F}}$, e $\gamma$ uma curva fechada,
		\begin{equation*}
			\oint_{\gamma}\vec{\boldsymbol{F}}\cdot d\gamma=0.
		\end{equation*}
	\end{exemplo}

	\begin{exemplo}
		\textbf{(Campo não conservativo irrotacional)}
		
		Seja
		\begin{equation*}
			\vec{\boldsymbol{F}}=-\frac{y}{x^2+y^2}\,\mathbf{i}+\frac{x}{x^2+y^2}\,\mathbf{j},\quad(x,y)\neq (0,0)
		\end{equation*}
		Verificando a condição de irrotacionalidade, obtêm-se
		\begin{align*}
			\frac{\partial P}{\partial y}=\frac{\partial}{\partial y}\left(-\frac{y}{x^2+y^2}\right)&=\frac{\partial}{\partial y}(-y(x^2+y^2)^{-1})\\
			&=\frac{\partial}{\partial y}(-y)\cdot(x^2+y^2)^{-1}+\frac{\partial}{\partial y}(x^2+y^2)^{-1}\cdot(-y)\\
			&=-(x^2+y^2)^{-1}-y\frac{\partial}{\partial u}u^{-1}\cdot \frac{\partial}{\partial y}(x^2+y^2),\quad u=x^2+y^2\\
			&=-\frac{1}{x^2+y^2}+\frac{2y^2}{(x^2+y^2)^2}=\frac{-(x^2+y^2)+2y^2}{(x^2+y^2)^2}\\
			&=\frac{y^2-x^2}{(x^2+y^2)^2}
		\end{align*}
		e
		\begin{align*}
			\frac{\partial Q}{\partial x}=\frac{\partial}{\partial x}\left(\frac{x}{x^2+y^2}\right)&=\frac{\partial}{\partial x}(x(x^2+y^2)^2)\\
			&=\frac{\partial}{\partial x}x\cdot (x^2+y^2)^{-1}+\frac{\partial}{\partial y}(x^2+y^2)^{-1}\cdot x\\
			&=\frac{1}{x^2+y^2}-\frac{\partial}{\partial u}u^{-1}\cdot \frac{\partial}{\partial x}(x^2+y^2)\cdot x,\quad u=x^2+y^2\\
			&=\frac{1}{x^2+y^2}-\frac{2x^2}{(x^2+y^2)^2}\\
			&=\frac{y^2-x^2}{(x^2+y^2)^2}
		\end{align*}
		Dessa forma,
		\begin{equation*}
			\frac{\partial P}{\partial y}=\frac{\partial Q}{\partial x}=\frac{y^2-x^2}{(x^2+y^2)^2},\quad \operatorname{rot}\vec{\boldsymbol{F}}=0.
		\end{equation*}
		Considerando a curva $\gamma(t)=(\cos t,\sin t)$, $t\in [0,2\pi]$,
		\begin{align*}
			\int_{\gamma}\vec{\boldsymbol{F}}\cdot d\gamma&=\int_{\gamma}-\frac{y}{x^2+y^2}\,dx+\frac{x}{x^2+y^2}\,dy\\
			&=-y\int_{\gamma(0)=(1,0)}^{\gamma(1)=(\cos 1,\sin 1)}\frac{1}{x^2+y^2}\, dx+x\int_{\gamma(0)=(1,0)}^{\gamma(1)=(\cos 1,\sin 1)}\frac{1}{x^2+y^2}\,dy\\
			&=\frac{2}{3}yx^3-yx^3-xy^3\Big|_{\gamma(0)=(1,0)}^{\gamma(1)=(\cos 1,\sin 1)}+xy^3+yx^3-\frac{2}{3}xy^3\Big|_{\gamma(0)=(1,0)}^{\gamma(1)=(\cos 1,\sin 1)},
		\end{align*}
		em que
		\begin{equation*}
			\oint_{\gamma}\vec{\boldsymbol{F}}\cdot d\gamma\neq 0.
		\end{equation*}
	\end{exemplo}
